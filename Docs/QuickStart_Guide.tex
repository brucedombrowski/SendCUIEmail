\section*{Before You Begin: Choose the Right Method}

\colorbox{lightgreen}{\parbox{\dimexpr\textwidth-2\fboxsep}{
\textbf{OPTION 1: S/MIME Only (Both Parties Have PIV/CAC)}\\[3pt]
If you have your recipient's S/MIME certificate, you can use S/MIME alone:
\begin{enumerate}[leftmargin=0.3in, topsep=2pt, itemsep=0pt]
    \item Compose your email in Outlook with CUI files attached
    \item Click \textbf{Sign} and \textbf{Encrypt} before sending
\end{enumerate}
}}

\vspace{6pt}

\colorbox{cuiyellow}{\parbox{\dimexpr\textwidth-2\fboxsep}{
\textbf{OPTION 2: S/MIME + SendCUIEmail (Maximum Protection)}\\[3pt]
For defense-in-depth, use \textbf{both} S/MIME and SendCUIEmail together:
\begin{itemize}[leftmargin=0.3in, topsep=2pt, itemsep=0pt]
    \item Files are encrypted with a password (content protection)
    \item Email is signed and encrypted with S/MIME (transport protection)
    \item Password sent separately (out-of-band delivery)
\end{itemize}
\textbf{Why?} If email is accidentally forwarded or misdirected, files remain encrypted. Separating password from content minimizes accidental leakage.
}}

\vspace{6pt}

\textbf{OPTION 3: SendCUIEmail Only} -- Use when recipient lacks PKI capability (no PIV/CAC/S/MIME), such as external partners or contractors.

\textbf{TIP:} Always \textbf{sign your email} if you have a PIV/CAC -- this proves it came from you.

\section*{Part A: Encrypting Files (Sender)}

\colorbox{lightblue}{\parbox{\dimexpr\textwidth-2\fboxsep}{\textbf{Step 1: Launch the Tool}}}
\begin{enumerate}[leftmargin=0.4in, topsep=2pt]
    \item Navigate to the SendCUIEmail folder
    \item Double-click \texttt{Encrypt.bat}
    \item A blue PowerShell window will open
\end{enumerate}

\colorbox{lightblue}{\parbox{\dimexpr\textwidth-2\fboxsep}{\textbf{Step 2: Select Files}}}
\begin{enumerate}[leftmargin=0.4in, topsep=2pt]
    \item When prompted, drag files into the window OR type the file path
    \item Press Enter when done adding files
\end{enumerate}

\colorbox{lightblue}{\parbox{\dimexpr\textwidth-2\fboxsep}{\textbf{Step 3: Choose CUI Category}}}
\begin{enumerate}[leftmargin=0.4in, topsep=2pt]
    \item Select the appropriate CUI marking (0-4)
    \item Press Enter for basic CUI if unsure
\end{enumerate}

\colorbox{lightblue}{\parbox{\dimexpr\textwidth-2\fboxsep}{\textbf{Step 4: Create Password}}}
\begin{enumerate}[leftmargin=0.4in, topsep=2pt]
    \item Type a password (minimum 8 characters) OR press Enter for auto-generated passphrase
    \item \textbf{IMPORTANT:} Characters will NOT appear as you type -- this is normal!
    \item If using auto-generated passphrase, copy it down immediately
    \item Otherwise, type your password again to confirm
    \item Remember this password/passphrase -- you'll share it separately
\end{enumerate}

\colorbox{lightblue}{\parbox{\dimexpr\textwidth-2\fboxsep}{\textbf{Step 5: Send the Files}}}
\begin{enumerate}[leftmargin=0.4in, topsep=2pt]
    \item Find the output folder (path shown in yellow)
    \item Attach the \texttt{.Locked} files and \texttt{README.md} to your email
    \item If you have a PIV/CAC, click \textbf{Sign} before sending (proves it's from you)
    \item Send the email to your recipient
\end{enumerate}

\vspace{6pt}
\colorbox{lightred}{\parbox{\dimexpr\textwidth-2\fboxsep}{
\textbf{WARNING: Password Delivery}\\[3pt]
\textbf{COMPLIANT:} Phone call, SMS text, in-person, or S/MIME encrypted email\\
\textbf{NOT COMPLIANT:} Unencrypted email (even if separate from files)
}}

\section*{Part B: Decrypting Files (Recipient)}

\colorbox{lightblue}{\parbox{\dimexpr\textwidth-2\fboxsep}{\textbf{Step 1: Save the Files}}}
\begin{enumerate}[leftmargin=0.4in, topsep=2pt]
    \item Download all \texttt{.Locked} files from the email
    \item Download the \texttt{README.md} file
    \item Save them all to the same folder (e.g., Desktop)
\end{enumerate}

\colorbox{lightblue}{\parbox{\dimexpr\textwidth-2\fboxsep}{\textbf{Step 2: Open PowerShell}}}
\begin{enumerate}[leftmargin=0.4in, topsep=2pt]
    \item Press \textbf{Windows Key + X}
    \item Click ``Windows PowerShell'' or ``Terminal''
    \item A blue command window will open
\end{enumerate}

\colorbox{lightblue}{\parbox{\dimexpr\textwidth-2\fboxsep}{\textbf{Step 3: Run the Decryption Command}}}
\begin{enumerate}[leftmargin=0.4in, topsep=2pt]
    \item Open the \texttt{README.md} file (double-click or use Notepad)
    \item Find the ``PowerShell One-Liner'' section
    \item Copy the entire command (starts with \texttt{\$f=Read-Host})
    \item Paste it into PowerShell and press Enter
\end{enumerate}

\colorbox{lightblue}{\parbox{\dimexpr\textwidth-2\fboxsep}{\textbf{Step 4: Enter File Path and Password}}}
\begin{enumerate}[leftmargin=0.4in, topsep=2pt]
    \item When prompted for ``File'', type the full path to a \texttt{.Locked} file\\
          \textit{Example:} \texttt{C:\textbackslash Users\textbackslash YourName\textbackslash Desktop\textbackslash Document.pdf.Locked}
    \item When prompted for ``Password'', type the password you received
    \item \textbf{IMPORTANT:} Characters will NOT appear -- this is normal!
\end{enumerate}

\colorbox{lightblue}{\parbox{\dimexpr\textwidth-2\fboxsep}{\textbf{Step 5: Verify Decryption}}}
\begin{enumerate}[leftmargin=0.4in, topsep=2pt]
    \item The decrypted file appears in the same folder
    \item \textit{Example:} \texttt{Document.pdf.Locked} becomes \texttt{Document.pdf}
    \item Repeat steps 3-4 for each \texttt{.Locked} file
\end{enumerate}

\vspace{12pt}
\colorbox{lightgreen}{\parbox{\dimexpr\textwidth-2\fboxsep}{
\textbf{TIP:} Use the same password for all files in a batch.\\
If decryption fails, double-check the password and file path.
}}
